<!-- Modal -->
<div id="modal" class="modal">
  <span class="close" onclick="closeModal()">&times;</span>
  <div class="modal-content">
    <img id="modal-image" src="" alt="">
    <a class="prev" onclick="changeSlide(-1)">&#10094;</a>
    <a class="next" onclick="changeSlide(1)">&#10095;</a>
  </div>
</div>

<script>
  const images = Array.from(document.querySelectorAll(".gallery img"));
  const modal = document.getElementById("modal");
  const modalImage = document.getElementById("modal-image");
  let currentIndex = 0;

  // Handle smooth decode and fade-in for gallery images
  images.forEach((img, index) => {
    const wrapper = img.parentElement;

    function handleLoaded() {
      img.classList.add("loaded");
      if (wrapper) wrapper.classList.add("loaded");
    }

    if (img.complete) {
      handleLoaded();
    } else {
      img.addEventListener("load", handleLoaded);
    }

    img.addEventListener("click", () => {
      currentIndex = index;
      openModal();
    });
  });

  function openModal() {
    modal.style.display = "flex";
    // Trigger layout paint before adding transition opacity class
    requestAnimationFrame(() => {
      modal.classList.add("show");
    });
    showSlide(currentIndex);
  }

  function closeModal() {
    modal.classList.remove("show");
    setTimeout(() => {
      modal.style.display = "none";
      modalImage.src = "";
    }, 250);
  }

  function changeSlide(n) {
    currentIndex = (currentIndex + n + images.length) % images.length;
    showSlide(currentIndex);
  }

  function showSlide(index) {
    modalImage.style.opacity = "0.3";
    const targetImg = images[index];
    
    modalImage.src = targetImg.src;
    modalImage.alt = targetImg.alt;

    if (modalImage.complete) {
      modalImage.style.opacity = "1";
    } else {
      modalImage.onload = () => {
        modalImage.style.opacity = "1";
      };
    }

    // Preload adjacent images in the background for instant navigation
    const prevIdx = (index - 1 + images.length) % images.length;
    const nextIdx = (index + 1) % images.length;
    new Image().src = images[prevIdx].src;
    new Image().src = images[nextIdx].src;
  }

  modal.addEventListener("click", (e) => {
    if (!e.target.closest(".modal-content") && !e.target.closest(".prev") && !e.target.closest(".next")) {
      closeModal();
    }
  });

  document.addEventListener("keydown", (e) => {
    if (modal.style.display === "flex") {
      if (e.key === "ArrowLeft") changeSlide(-1);
      else if (e.key === "ArrowRight") changeSlide(1);
      else if (e.key === "Escape") closeModal();
    }
  });
</script>
